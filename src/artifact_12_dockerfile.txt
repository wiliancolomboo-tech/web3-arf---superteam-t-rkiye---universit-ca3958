FROM node:20-alpine AS runner

WORKDIR /usr/src/app

# Install system dependencies for build
RUN apk add --no-cache python3 make g++ git

COPY client/package*.json ./
RUN npm ci

COPY client/tsconfig.json ./
COPY client/src ./src

RUN npm run build

EXPOSE 8080
USER node

CMD ["node", "dist/index.js"]
