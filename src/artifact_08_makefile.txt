.PHONY: build test deploy-devnet start docker-up docker-down

build:
	anchor build

test:
	anchor test

deploy-devnet:
	solana config set --url devnet
	anchor build
	solana program deploy target/deploy/arf_settlement.so

start:
	npx ts-node src/relayer/settlementDaemon.ts simulate-event

docker-up:
	docker compose up -d --build

docker-down:
	docker compose down -v
